import React, { useEffect } from 'react';

export const SqlHighlighter = () => {
  useEffect(() => {
    const highlightSql = async () => {
      try {
        const shiki = await import('https://esm.sh/shiki@3.6.0');
        const sqlGrammarUrl = 'https://raw.githubusercontent.com/microsoft/vscode/main/extensions/sql/syntaxes/sql.tmLanguage.json';
        const sqlGrammarResponse = await fetch(sqlGrammarUrl);
        const sqlGrammar = await sqlGrammarResponse.json();
        const highlighter = await shiki.getHighlighter({
          themes: ['mint'],
          langs: [
            {
              id: 'sql',
              scopeName: 'source.sql',
              grammar: sqlGrammar,
              aliases: ['sql'],
            },
          ],
        });

        // Find all code elements that should be highlighted as SQL
        const codeElements = document.querySelectorAll('pre code.language-sql');

        codeElements.forEach(element => {
          const code = element.textContent;
          const parent = element.parentElement;
          // Replace the pre element's content with the highlighted version
          parent.outerHTML = highlighter.codeToHtml(code, { lang: 'sql', theme: 'github-dark' });
        });

      } catch (error) {
        console.error('Failed to highlight SQL:', error);
      }
    };

    highlightSql();
  }, []); // The empty dependency array ensures this runs once after the component mounts

  return null; // This component does not render anything itself
};

export default SqlHighlighter;
